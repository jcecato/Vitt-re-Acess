Divergência Crítica Detectada (Realtime DB vs. Firestore): O sistema atual usa Firebase Realtime Database (firebase/database), gravando os objetos patient e results (formatados para UI). Para que o app Vittáre Acess leia diretamente em /patients/{email}/anthropometry/{docId} via Firestore (ou mantenha um nó estruturado metrics), é necessário serializar os valores numéricos puros junto com os textos de saída final das referências dentro de um objeto unificado metrics.

export const numBR = (n: number | null | undefined, max = 2): string => {
  if (n === null || n === undefined || isNaN(n)) return '';
  return new Intl.NumberFormat('pt-BR', { maximumFractionDigits: max }).format(n);
};

export const parseDate = (dateString: string): Date | null => {
  if (!dateString) return null;
  const parts = dateString.split('-');
  return new Date(parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2]));
};

export const calcularIdade = (nasc: Date): number => {
  const hoje = new Date();
  let idade = hoje.getFullYear() - nasc.getFullYear();
  const m = hoje.getMonth() - nasc.getMonth();
  if (m < 0 || (m === 0 && hoje.getDate() < nasc.getDate())) idade--;
  return idade;
};

export const generateTimestampId = (): string => {
  const now = new Date();
  const pad = (n: number) => String(n).padStart(2, '0');
  return `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}-${pad(now.getHours())}-${pad(now.getMinutes())}-${pad(now.getSeconds())}`;
};

const calcularCMB = (bracoCm: number, dobraTricipitalMm: number): number => {
  let resultado = bracoCm - (0.314 * dobraTricipitalMm);
  return parseFloat(resultado.toFixed(1));
};

const calcFFMI = (peso: number, gorduraPct: number, alturaCm: number): number | null => {
  const h = (alturaCm || 0) / 100;
  if (!peso || !h) return null;
  const massaMagra = peso * (1 - (Number(gorduraPct || 0) / 100));
  return massaMagra > 0 ? massaMagra / (h * h) : null;
};

const getRef = (type: string, sex: string, value: number): RefItem => {
  const list = ANTROPOMETRIA_REF[type][sex] || ANTROPOMETRIA_REF[type]['ALL'];
  if (!list) return { label: 'N/A', color: '#333', status: 'default', max: 0 };
  for (const item of list) {
    if (value <= item.max) return item;
  }
  return list[list.length - 1];
};

const getFatRef = (sex: string, age: number): any => {
  if (!REFERENCIAS_GORDURA[sex]) return null;
  let ref = REFERENCIAS_GORDURA[sex].find((r: any) => age >= r.minAge && age <= r.maxAge);
  if (!ref) {
    if (age < 20) ref = REFERENCIAS_GORDURA[sex][0];
    else ref = REFERENCIAS_GORDURA[sex][REFERENCIAS_GORDURA[sex].length - 1];
  }
  return ref;
};

const obterClassificacaoMusculo = (sexo: string, idade: number, valorAtual: number): any => {
  const faixaEtaria = (idade >= 60) ? 'idoso' : 'adulto';
  const referencias = REFERENCIAS_MUSCULO[faixaEtaria][sexo];
  const classificacao = referencias.find((ref: any) => valorAtual <= ref.max) || referencias[referencias.length - 1];
  const referenciaIdeal = referencias.find((ref: any) => ref.label.includes("Normal"));
  return {
    status: classificacao.label,
    cor: classificacao.cor,
    textoIdeal: referenciaIdeal ? referenciaIdeal.ideal : "Consulte Tabela",
    referencias: referencias
  };
};

const createBox = (label: string, value: number, unit: string, range: number[], isPercent: boolean, suffix = ''): ReportBox => {
  const [mn, mx] = range;
  const margin = isPercent ? 0 : 10;
  const maxScale = isPercent ? 100 : Math.max(mx + margin, value);
  const lw = (mn / maxScale) * 100;
  const mw = ((mx - mn) / maxScale) * 100;
  const rw = isPercent ? Math.max(0, 100 - mx) : Math.max(0, 100 - (lw + mw));
  const mk = Math.min((value / maxScale) * 100, 100);
  return {
    title: label,
    value: `${numBR(value)}${unit}${suffix}`,
    subtext: `Ideal: ${mn} – ${mx} ${unit}`,
    progress: {
      segments: [
        { l: 0, w: lw, c: '#e74c3c' },
        { l: lw, w: mw, c: '#1ca657' },
        { l: lw + mw, w: rw, c: '#e74c3c' }
      ],
      marker: mk
    }
  };
};

const createVisceralBox = (val: number): ReportBox => {
  const max = 20;
  const mk = Math.min((val / max) * 100, 100);
  return {
    title: 'Gordura Visceral',
    value: val.toString(),
    subtext: 'Normal: 1-9 | Alto: >10',
    progress: {
      segments: [
        { l: 0, w: 45, c: '#1ca657' },
        { l: 45, w: 25, c: '#f1c40f' },
        { l: 70, w: 30, c: '#e74c3c' }
      ],
      marker: mk
    }
  };
};

const createMuscleBox = (pctVal: number, kgVal: number, classData: any): ReportBox => {
  const refs = classData.referencias;
  const maxScale = refs[2].max + 10;
  const mk = Math.min((pctVal / maxScale) * 100, 100);
  const w1 = (refs[0].max / maxScale) * 100;
  const w2 = ((refs[1].max - refs[0].max) / maxScale) * 100;
  const w3 = ((refs[2].max - refs[1].max) / maxScale) * 100;
  const w4 = 100 - w1 - w2 - w3;
  return {
    title: 'Massa Muscular',
    value: `${numBR(pctVal)}% – ${numBR(kgVal)} kg`,
    subtext: `Ideal: ${classData.textoIdeal}`,
    progress: {
      segments: [
        { l: 0, w: w1, c: refs[0].cor },
        { l: w1, w: w2, c: refs[1].cor },
        { l: w1 + w2, w: w3, c: refs[2].cor },
        { l: w1 + w2 + w3, w: w4, c: refs[3].cor }
      ],
      marker: mk
    }
  };
};

const createMuacBox = (val: number, sex: string, ref: RefItem): ReportBox => {
  const maxScale = 50;
  const mk = Math.min((val / maxScale) * 100, 100);
  let redLim, yellowLim;
  if (sex === 'M') {
    redLim = 23; yellowLim = 27;
  } else {
    redLim = 22; yellowLim = 26;
  }
  const wRed = (redLim / maxScale) * 100;
  const wYellow = ((yellowLim - redLim) / maxScale) * 100;
  const wGreen = 100 - wRed - wYellow;
  return {
    title: 'Braço (MUAC)',
    value: `${numBR(val)} cm`,
    subtext: ref.label,
    progress: {
      segments: [
        { l: 0, w: wRed, c: '#d93025' },
        { l: wRed, w: wYellow, c: '#f1c40f' },
        { l: wRed + wYellow, w: wGreen, c: '#1ca657' }
      ],
      marker: mk
    }
  };
};

export const generateReportData = (data: PatientData): FullReport => {
  const peso = parseFloat(data.peso) || 0;
  const altura = parseFloat(data.altura) || 0;
  const nascDate = parseDate(data.nascimento);
  const idade = nascDate ? calcularIdade(nascDate) : null;
  const sexo = data.sexo;
  const isElderly = idade !== null && idade >= 60;

  const res: any = { metas: [], bio: [], mmlg: [], circ: [], gasto: [], checklist: [], idade };
  const checklist: ChecklistItem[] = [];
  let cmbData = null;

  if (altura > 0 && peso > 0) {
    const pGordura = parseFloat(data.gordura);
    const h2 = Math.pow((altura / 100), 2);
    const IMC_MIN = 22; const IMC_MAX = 24;
    const pesoIMCmin = IMC_MIN * h2;
    const pesoIMCmax = IMC_MAX * h2;
    let legenda = `Critério híbrido: IMC ${IMC_MIN}–${IMC_MAX}`;
    let corValor = '#1ca657';
    let pesoIdealMin = pesoIMCmin;
    let pesoIdealMax = pesoIMCmax;

    if (!isNaN(pGordura) && sexo) {
      const massaMagra = peso * (1 - pGordura / 100);
      const alvo = METAS_GORDURA[sexo];
      const gMin = alvo?.saude?.[0];
      const gMax = alvo?.saude?.[1];
      if (typeof gMin === 'number' && typeof gMax === 'number') {
        if (pGordura > gMin) {
          const pesoCompMin = massaMagra / (1 - gMin / 100);
          const pesoCompMax = massaMagra / (1 - gMax / 100);
          const interMin = Math.max(pesoIMCmin, pesoCompMin);
          const interMax = Math.min(pesoIMCmax, pesoCompMax);
          if (interMin <= interMax) {
            pesoIdealMin = interMin;
            pesoIdealMax = interMax;
            legenda += ` + %G alvo (${gMin}–${gMax}%)`;
          } else {
            legenda += ` (sem intersecção com %G alvo; priorizado IMC)`;
            corValor = '#f1c40f';
          }
        } else {
          legenda += `; %G atual ≤ ${gMin}% — foco em recomposição (↑MM, peso estável)`;
        }
      }
    }

    res.metas.push({
      title: 'Peso Ideal',
      value: `${numBR(pesoIdealMin, 1)} – ${numBR(pesoIdealMax, 1)} kg`,
      subtext: legenda,
      complement: DESCRICOES_COMPLEMENTARES.pesoIdeal,
      valueColor: corValor
    });

    const ffmi = calcFFMI(peso, parseFloat(data.gordura), altura);
    if (ffmi) {
      const statusFFMI = ffmi < 18 ? 'baixo' : (ffmi <= 21 ? 'adequado' : 'elevado');
      const corFFMI = ffmi < 18 ? '#d93025' : (ffmi <= 21 ? '#1ca657' : '#0044cc');
      
      const maxScale = 30;
      const mk = Math.min((ffmi / maxScale) * 100, 100);
      const wRed = (18 / maxScale) * 100;
      const wGreen = ((21 - 18) / maxScale) * 100;
      const wBlue = 100 - wRed - wGreen;

      res.metas.push({
        title: 'FFMI (Índice de Massa Magra)',
        value: `${numBR(ffmi, 1)} kg/m²`,
        subtext: `Referência: 18–21 (adultos) — ${statusFFMI}`,
        complement: DESCRICOES_COMPLEMENTARES.ffmi,
        valueColor: corFFMI,
        progress: {
          segments: [
            { l: 0, w: wRed, c: '#d93025' }, // Baixo
            { l: wRed, w: wGreen, c: '#1ca657' }, // Adequado
            { l: wRed + wGreen, w: wBlue, c: '#0044cc' } // Elevado
          ],
          marker: mk
        }
      });
    }
  }

  const h = altura / 100;
  const imc = (peso && altura) ? parseFloat((peso / (h * h)).toFixed(2)) : null;
  let imcClass = { key: '', label: 'Indisponível' };

  if (imc) {
    if (isElderly) {
      if (imc < 22) imcClass = { key: 'magreza', label: 'Baixo Peso (<22.0)' };
      else if (imc <= 27) imcClass = { key: 'normal', label: 'Eutrofia (22.0–27.0)' };

      else if (imc < 25) imcClass = { key: 'normal', label: 'Adequado (18.5–24.9)' };

      else if (imc < 35) imcClass = { key: 'obesidade1', label: 'Obesidade 1 (30.0–34.9)' };
      else if (imc < 40) imcClass = { key: 'obesidade2', label: 'Obesidade 2 (35.0–39.9)' };
      else imcClass = { key: 'obesidade3', label: 'Obesidade 3 (>=40.0)' };
    }

    checklist.push({
      label: 'IMC',
      status: imcClass.key === 'normal'
        ? 'success'
        : (imcClass.key.includes('obesidade') || imcClass.key === 'magreza' || imcClass.key === 'baixo_peso'
          ? 'danger' : 'warning'),
      value: imcClass.label
    });
  }

  const pGordura = parseFloat(data.gordura);
  if (!isNaN(pGordura)) {

  const visceral = parseFloat(data.visceral);

      label: 'Gordura Visceral',
      status: visceral <= 9 ? 'success' : (visceral <= 14 ? 'warning' : 'danger'),

      value: rcq.toFixed(2),
      subtext: ref.label,
      complement: DESCRICOES_COMPLEMENTARES.rcq[ref.status],
      valueColor: ref.color
    });
    checklist.push({
      label: 'RCQ (Risco Metabólico)',
      status: toChecklistStatus(ref),
      value: ref.label
    });
  }

  if (cintura && altura) {
    const rce = cintura / altura;
    const ref = getRef('rce', 'ALL', rce);
    
    const maxScale = 0.8;
    const mk = Math.min((rce / maxScale) * 100, 100);
    const wYellow = (0.40 / maxScale) * 100;
    const wGreen = ((0.50 - 0.40) / maxScale) * 100;
    const wRed = 100 - wYellow - wGreen;

    res.circ.push({
      title: 'Relação Cintura-Estatura',
      value: rce.toFixed(2),
      subtext: ref.label,
      complement: DESCRICOES_COMPLEMENTARES.rce[ref.status],
      valueColor: ref.color,
      progress: {
        segments: [
          { l: 0, w: wYellow, c: '#f1c40f' }, // Baixo
          { l: wYellow, w: wGreen, c: '#1ca657' }, // Adequado
          { l: wYellow + wGreen, w: wRed, c: '#d93025' } // Alto
        ],
        marker: mk
      }
    });
  }

  const braco = parseFloat(data.braco);
  const dobraTri = parseFloat(data.dobraTricipital);
  const coxa = parseFloat(data.coxa);
  const dobraCoxa = parseFloat(data.dobraCoxa);

  if (braco && sexo) {
    const refMUAC = getRef('muac', sexo, braco);
    res.circ.push({
      ...createMuacBox(braco, sexo, refMUAC),
      complement: DESCRICOES_COMPLEMENTARES.muac[refMUAC.status]
    });

    checklist.push({
      label: 'Braço (MUAC)',
      status: toChecklistStatus(refMUAC),
      value: refMUAC.label
    });

    if (!isNaN(dobraTri)) {
      const cmb = calcularCMB(braco, dobraTri);
      const refCMB = getRef('cmb', sexo, cmb);
      cmbData = { value: refCMB.label, status: toChecklistStatus(refCMB) };
      res.circ.push({
        title: 'Reserva Muscular Real (CMB)',
        value: `${numBR(cmb)} cm`,
        subtext: refCMB.label,
        complement: DESCRICOES_COMPLEMENTARES.cmb[refCMB.status],
        valueColor: refCMB.color
      });

      const refTSF = getRef('tsf', sexo, dobraTri);
      res.circ.push({
        title: 'Dobra Tricipital',
        value: `${dobraTri} mm`,
        subtext: refTSF.label,
        complement: DESCRICOES_COMPLEMENTARES.dobraTricipital[refTSF.status],
        valueColor: refTSF.color
      });
    }
  }

  const panturrilha = parseFloat(data.panturrilha);
  if (panturrilha) {
    let subtext = 'Adequado';
    let color = '#1ca657';
    let status = 'normal';
    let chkStatus = 'success';
    if (panturrilha < 31) {
      subtext = 'Baixo';
      color = '#d93025';
      status = 'low';
      chkStatus = 'danger';
    } else if ((sexo === 'M' && panturrilha < 34) || (sexo === 'F' && panturrilha < 33)) {
      subtext = 'Baixo';
      color = '#f1c40f';
      status = 'low';
      chkStatus = 'warning';
    } else if ((sexo === 'M' && panturrilha > 40) || (sexo === 'F' && panturrilha > 39)) {
      subtext = 'Alto';
      color = '#d93025';
      status = 'high';
      chkStatus = 'danger';
    }
    res.circ.push({
      title: 'Panturrilha',
      value: `${numBR(panturrilha)} cm`,
      subtext: subtext,
      complement: DESCRICOES_COMPLEMENTARES.panturrilha[status],
      valueColor: color
    });
    checklist.push({
      label: 'Risco Sarcopenia',
      status: chkStatus as any,
      value: subtext
    });
  }

  if (!isNaN(coxa) && coxa > 0 && !isNaN(dobraCoxa) && dobraCoxa > 0) {
    const dobraCoxaCm = dobraCoxa / 10;
    const areaTotal = (coxa * coxa) / (4 * Math.PI);
    const circMuscular = coxa - Math.PI * dobraCoxaCm;
    const circMuscularCorrigida = Math.max(circMuscular, 0.1);
    const areaMuscular = (circMuscularCorrigida * circMuscularCorrigida) / (4 * Math.PI);
    const areaGordura = Math.max(areaTotal - areaMuscular, 0);
    const pctMusculo = areaMuscular / areaTotal * 100;
    const pctGordura = areaGordura / areaTotal * 100;

    let labelReserva = 'Adequado';
    let color = '#1ca657';
    let status = 'normal';
    let chkStatus = 'success';

    const limitLow = (idade !== null && idade >= 60) ? 50 : 55;
    const limitHighFat = (idade !== null && idade >= 60) ? 40 : 45;

    if (pctMusculo < limitHighFat) {
      labelReserva = 'Alto (Excesso de Gordura)';
      color = '#d93025';
      status = 'high';
      chkStatus = 'danger';
    } else if (pctMusculo < limitLow) {
      labelReserva = 'Baixo (Déficit de Músculo)';
      color = '#f1c40f';
      status = 'low';
      chkStatus = 'warning';
    } else {
      labelReserva = 'Adequado';
      color = '#1ca657';
      status = 'normal';
      chkStatus = 'success';
    }

    res.circ.push({
      title: 'Composição média de coxa',
      value: `${numBR(pctMusculo, 1)}% músculo* | ${numBR(pctGordura, 1)}% gordura`,
      subtext: `${labelReserva}. Estimativa geométrica.`,
      complement: DESCRICOES_COMPLEMENTARES.compCoxa[status],
      valueColor: color
    });

    let gorduraCoxaStatus = 'normal';
    let gorduraCoxaLabel = 'Adequado';
    let gorduraCoxaColor = '#1ca657';
    
    if (dobraCoxa < 10) {
      gorduraCoxaStatus = 'low';
      gorduraCoxaLabel = 'Baixo';
      gorduraCoxaColor = '#f1c40f';
    } else if (dobraCoxa > 30) {
      gorduraCoxaStatus = 'high';
      gorduraCoxaLabel = 'Alto';
      gorduraCoxaColor = '#d93025';
    }

    res.circ.push({
      title: 'Gordura subcutânea de coxa',
      value: `${numBR(dobraCoxa, 1)} mm`,
      subtext: gorduraCoxaLabel,
      complement: DESCRICOES_COMPLEMENTARES.gorduraCoxa[gorduraCoxaStatus],
      valueColor: gorduraCoxaColor
    });

    checklist.push({
      label: 'Reserva muscular de coxa',
      status: chkStatus as any,
      value: labelReserva
    });

    if (cmbData) {
      checklist.push({
        label: 'Reserva Muscular Real (CMB)',
        status: cmbData.status as any,
        value: cmbData.value
      });
    }
  } else {
    if (cmbData) {
      checklist.push({
        label: 'Reserva Muscular Real (CMB)',
        status: cmbData.status as any,
        value: cmbData.value
      });
    }
  }

  ['coxa', 'antebraco', 'punho', 'umero'].forEach(k => {
    const v = parseFloat((data as any)[k]);
    if (!isNaN(v)) {
      res.circ.push({
        title: k.charAt(0).toUpperCase() + k.slice(1),
        value: `${numBR(v)} cm`,
        subtext: 'Medida aferida'
      });
    }
  });

  res.checklist = checklist;
  return { patient: data, results: res };
};

export const openPrintWindow = (reportHtml: string) => {
    try {
      const win = window.open('', '_blank');
      if (win) {
        win.document.open();
        win.document.write(reportHtml);
        win.document.close();
        win.focus();
        setTimeout(() => {
          try { win.print(); } catch (e) { console.error('Erro ao acionar impressão na janela:', e); }
        }, 300);
        return;
      }
    } catch (err) {
      console.warn('Pop-up bloqueado, utilizando fallback de iframe oculto para impressão:', err);
    }

    // Fallback: Iframe invisível para impressão segura sem pop-up
    const existingFrame = document.getElementById('print-sandbox-frame');
    if (existingFrame) existingFrame.remove();

    const iframe = document.createElement('iframe');
    iframe.id = 'print-sandbox-frame';
    iframe.style.position = 'fixed';
    iframe.style.right = '0';
    iframe.style.bottom = '0';
    iframe.style.width = '0';
    iframe.style.height = '0';
    iframe.style.border = '0';
    document.body.appendChild(iframe);

    const doc = iframe.contentWindow?.document;
    if (doc) {
      doc.open();
      doc.write(reportHtml);
      doc.close();
      setTimeout(() => {
        iframe.contentWindow?.focus();
        iframe.contentWindow?.print();
      }, 400);
    }
};
